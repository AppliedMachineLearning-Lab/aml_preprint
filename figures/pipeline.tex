% Example TikZ figure: selective prediction pipeline.
% Requires \usetikzlibrary{arrows.meta,positioning,shapes.geometric} in the preamble.
% Colours are from the Tol muted palette defined in amlpreprint.sty.
\begin{tikzpicture}[
    node distance=6mm and 9mm,
    >={Stealth[length=2.2mm]},
    every node/.style={font=\small},
    box/.style={draw=#1, fill=#1!25, line width=0.8pt, rounded corners=2pt,
                minimum height=9mm, minimum width=22mm, align=center},
    decision/.style={draw=tolindigo, fill=tolindigo!15, line width=0.8pt,
                     diamond, aspect=2, inner sep=1pt, align=center},
    arrow/.style={->, line width=0.8pt, draw=black!70},
]
  \node[box=tolcyan]                       (input)   {Input $\vx$};
  \node[box=tolsand, right=of input]       (model)   {Classifier\\$f(\vx; \vtheta)$};
  \node[box=tololive, right=of model]      (conf)    {Confidence\\$c(\vx)$};
  \node[decision, right=of conf]           (thresh)  {$c(\vx) \geq \tau$?};
  \node[box=tolteal, above right=0mm and 9mm of thresh.north east, anchor=south west]
                                           (answer)  {Answer $\hat{y}$};
  \node[box=tolrose, below right=0mm and 9mm of thresh.south east, anchor=north west]
                                           (abstain) {Abstain};

  \draw[arrow] (input) -- (model);
  \draw[arrow] (model) -- (conf);
  \draw[arrow] (conf)  -- (thresh);
  \draw[arrow] (thresh.north) |- node[pos=0.75, above] {yes} (answer.west);
  \draw[arrow] (thresh.south) |- node[pos=0.75, below] {no}  (abstain.west);
\end{tikzpicture}
